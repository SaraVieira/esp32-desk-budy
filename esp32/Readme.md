Create a `src/secrets.h` with:

```c++
#ifndef SECRETS_H
#define SECRETS_H

// Spotify API Credentials
#define SPOTIFY_CLIENT_ID "client-id"
#define SPOTIFY_CLIENT_SECRET "secret"
// create with https://spotify-refresh-token-generator.netlify.app/#welcome
#define SPOTIFY_REFRESH_TOKEN "refresh"

// wifi credentials
#define WIFI_SSID ""
#define WIFI_PASSWORD "choo choo"

#endif // SECRETS_H
```
